CREATE TABLE Borrower (
    Roll_no INT PRIMARY KEY,
    Name VARCHAR(50),
    DateOfIssue DATE,
    NameOfBook VARCHAR(100),
    Status CHAR(1)       
);

CREATE TABLE Fine (
    Roll_no INT,
    Date DATE,
    Amt INT
);

INSERT INTO Borrower VALUES
(101, 'Amit', '2025-10-01', 'DBMS', 'I'),
(102, 'Riya', '2025-09-15', 'JAVA', 'I'),
(103, 'Neha', '2025-10-25', 'OS', 'I');

DELIMITER $$

CREATE PROCEDURE calc_fine(IN rno INT, IN bname VARCHAR(50))
BEGIN
    DECLARE issueDate DATE;
    DECLARE days INT;
    DECLARE fine INT DEFAULT 0;

    SELECT DateOfIssue INTO issueDate
    FROM Borrower
    WHERE Roll_no = rno AND NameOfBook = bname;

    SET days = DATEDIFF(CURDATE(), issueDate);


    IF days > 30 THEN
        SET fine = days * 50;
    ELSEIF days > 15 THEN
        SET fine = days * 5;
    END IF;


    UPDATE Borrower SET Status = 'R'
    WHERE Roll_no = rno AND NameOfBook = bname;


    IF fine > 0 THEN
        INSERT INTO Fine VALUES (rno, CURDATE(), fine);
    END IF;

    SELECT rno AS Roll_No, bname AS Book, days AS Days_Late, fine AS Fine_Amount;
END$$

DELIMITER ;

CALL calc_fine(101, 'DBMS');

SELECT * FROM Fine;

CREATE TABLE Stud_Marks (
    Roll INT PRIMARY KEY AUTO_INCREMENT,
    Name VARCHAR(50),
    Total_Marks INT
);

CREATE TABLE Result (
    Roll INT,
    Name VARCHAR(50),
    Class VARCHAR(30)
);


INSERT INTO Stud_Marks (Name, Total_Marks) VALUES
('Amit', 1200),
('Riya', 950),
('Neha', 870),
('Karan', 800);

DELIMITER $$

CREATE PROCEDURE proc_Grade()
BEGIN
    DECLARE done INT DEFAULT 0;
    DECLARE sRoll INT;
    DECLARE sName VARCHAR(50);
    DECLARE sMarks INT;
    DECLARE grade VARCHAR(30);

    DECLARE stud_cursor CURSOR FOR
        SELECT Roll, Name, Total_Marks FROM Stud_Marks;

    DECLARE CONTINUE HANDLER FOR NOT FOUND SET done = 1;


    DELETE FROM Result;

    OPEN stud_cursor;

    read_loop: LOOP
        FETCH stud_cursor INTO sRoll, sName, sMarks;
        IF done THEN
            LEAVE read_loop;
        END IF;


        IF sMarks BETWEEN 990 AND 1500 THEN
            SET grade = 'Distinction';
        ELSEIF sMarks BETWEEN 900 AND 989 THEN
            SET grade = 'First Class';
        ELSEIF sMarks BETWEEN 825 AND 899 THEN
            SET grade = 'Higher Second Class';
        ELSE
            SET grade = 'Fail';
        END IF;
        INSERT INTO Result VALUES (sRoll, sName, grade);
    END LOOP;

    CLOSE stud_cursor;
END$$
DELIMITER ;
CALL proc_Grade();

SELECT * FROM Result;

DELIMITER $$

CREATE PROCEDURE merge_rollcall()
BEGIN
    DECLARE done INT DEFAULT 0;
    DECLARE n_roll INT;
    DECLARE n_name VARCHAR(50);

    -- Parameterized cursor (takes roll as input)
    DECLARE cur CURSOR FOR 
        SELECT roll, name FROM N_RollCall;

    DECLARE CONTINUE HANDLER FOR NOT FOUND SET done = 1;

    OPEN cur;
    read_loop: LOOP
        FETCH cur INTO n_roll, n_name;
        IF done THEN
            LEAVE read_loop;
        END IF;

        -- If roll not already in O_RollCall, insert it
        IF NOT EXISTS (SELECT * FROM O_RollCall WHERE roll = n_roll) THEN
            INSERT INTO O_RollCall VALUES (n_roll, n_name);
        END IF;
    END LOOP;
    CLOSE cur;
END$$

DELIMITER ;

CALL merge_rollcall();

SELECT * FROM O_RollCall;

    action_time TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

DELIMITER $$

CREATE TRIGGER before_borrower_update
BEFORE UPDATE ON borrower
FOR EACH ROW
BEGIN
    INSERT INTO library_audit (rollin, name, dateofissue, nameofbook, status, action_type)
    VALUES (OLD.rollin, OLD.name, OLD.dateofissue, OLD.nameofbook, OLD.status, 'UPDATE');
END$$

DELIMITER ;
DELIMITER $$

CREATE TRIGGER before_borrower_delete
BEFORE DELETE ON borrower
FOR EACH ROW
BEGIN
    INSERT INTO library_audit (rollin, name, dateofissue, nameofbook, status, action_type)
    VALUES (OLD.rollin, OLD.name, OLD.dateofissue, OLD.nameofbook, OLD.status, 'DELETE');
END$$

WHERE rollin = 1;
